CHAPTER 01

⚙️ Service Worker

A Service Worker is a background JavaScript file that sits between your app and the network. It powers offline mode, caching, and push notifications. Your HTML already has a basic inline version — this chapter upgrades it to a real, external file.

Step 1 — Why your current approach is limited

Your existing code registers a Service Worker from a Blob URL (lines 1353–1362). This works locally but has two problems: it can't intercept navigation requests correctly, and it can't receive push notifications because the browser blocks FCM on blob-origin workers.

Step 2 — Create sw.js at your project root

Create a new file called sw.js next to your HTML file with this content:

sw.js
// ─── sw.js — LASU Navigator Service Worker ───────────────────────────

const CACHE_NAME = 'lasu-nav-v2';
const STATIC_ASSETS = [
  '/',
  '/lasu-navigator-1.html',
  '/manifest.json',
  'https://fonts.googleapis.com/css2?family=Playfair+Display...',
  // add more assets here as you create them
];

// INSTALL — cache all static assets
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => cache.addAll(STATIC_ASSETS))
      .then(() => self.skipWaiting())  // activate immediately
  );
});

// ACTIVATE — delete old caches
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then(keys =>
      Promise.all(
        keys
          .filter(k => k !== CACHE_NAME)
          .map(k  => caches.delete(k))
      )
    ).then(() => self.clients.claim())
  );
});

// FETCH — Cache-first for static, Network-first for API
self.addEventListener('fetch', (event) => {
  const { request } = event;

  // Skip non-GET and cross-origin requests we don't cache
  if (request.method !== 'GET') return;

  // Firestore/Firebase requests → always go to network
  if (request.url.includes('firestore.googleapis.com') ||
      request.url.includes('firebase')) {
    event.respondWith(fetch(request));
    return;
  }

  // Everything else → Cache first, fallback to network
  event.respondWith(
    caches.match(request).then(cached => {
      const networkFetch = fetch(request).then(response => {
        const clone = response.clone();
        caches.open(CACHE_NAME).then(c => c.put(request, clone));
        return response;
      });
      return cached || networkFetch;
    })
  );
});

// PUSH — receive a push notification from FCM (Chapter 6)
self.addEventListener('push', (event) => {
  const data = event.data?.json() ?? {};
  event.waitUntil(
    self.registration.showNotification(
      data.title || 'LASU Navigator',
      {
        body: data.body || '',
        icon: '/icons/icon-192.png',
        badge: '/icons/badge-72.png',
        data: { url: data.url || '/' },
        actions: [{ action: 'open', title: 'Open App' }]
      }
    )
  );
});

// NOTIFICATIONCLICK — open app when notification is tapped
self.addEventListener('notificationclick', (event) => {
  event.notification.close();
  event.waitUntil(
    clients.openWindow(event.notification.data.url)
  );
});

🗄️ Firestore Database

Firestore is a NoSQL cloud database. Each student's profile, timetable, and grades are stored as documents and sync across devices automatically.

Step 1 — Database structure for LASU Navigator
Firestore Collections
students/ ← one doc per user

 {classId}/ → day, name, start, end, room

 {routeId}/ → lat, lng, routeName, lastSeen, status

 {id}/ → title, body, timestamp, type
Step 2 — Firestore security rules

In the Firebase Console, go to Firestore → Rules and paste:

Firestore Security Rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Students can only read/write their own profile
    match /students/{userId} {
      allow read, write: if request.auth != null
                         && request.auth.uid == userId;
    }

    // Timetable sub-collection — same rule
    match /students/{userId}/timetable/{classId} {
      allow read, write: if request.auth.uid == userId;
    }

    // Shuttles — any logged-in user can read; only admins write
    match /shuttles/{doc} {
      allow read: if request.auth != null;
      allow write: if request.auth.token.admin == true;
    }

    // Announcements — all logged-in users read; admins write
    match /announcements/{doc} {
      allow read: if request.auth != null;
      allow write: if request.auth.token.admin == true;
    }
  }
}

window.saveProfileToCloud = async (profileData) => {

Step 2 — IndexedDB wrapper for your timetable

Add this before your window.onload in your script:

In lasu-navigator-1.html <script>
// ── IndexedDB Helper ────────────────────────────────────────
const DB_NAME    = 'LasuNavDB';
const DB_VERSION = 1;
let   idb; // will hold the open database connection

window.openIDB = () => new Promise((resolve, reject) => {
  const req = indexedDB.open(DB_NAME, DB_VERSION);

  req.onupgradeneeded = (e) => {
    const db = e.target.result;
    // timetable store — keyed by id
    if (!db.objectStoreNames.contains('timetable'))
      db.createObjectStore('timetable', { keyPath: 'id', autoIncrement: true });
    // pending-queue — for offline writes
    if (!db.objectStoreNames.contains('pendingQueue'))
      db.createObjectStore('pendingQueue', { keyPath: 'id', autoIncrement: true });
  };

  req.onsuccess = e => { idb = e.target.result; resolve(idb); };
  req.onerror   = e => reject(e.target.error);
});

// ── IDB Helpers ──────────────────────────────────────────────
function idbAdd(store, data) {
  return new Promise((res, rej) => {
    const tx = idb.transaction(store, 'readwrite');
    const req = tx.objectStore(store).add(data);
    req.onsuccess = () => res(req.result);
    req.onerror   = () => rej(req.error);
  });
}

function idbGetAll(store) {
  return new Promise((res, rej) => {
    const tx = idb.transaction(store, 'readonly');
    const req = tx.objectStore(store).getAll();
    req.onsuccess = () => res(req.result);
    req.onerror   = () => rej(req.error);
  });
}

// ── OFFLINE QUEUE — queue writes when offline ────────────────
window.queueWrite = async (type, payload) => {
  await idbAdd('pendingQueue', {
    type, payload, timestamp: Date.now()
  });
};

// ── FLUSH QUEUE — call when back online ──────────────────────
window.addEventListener('online', async () => {
  showToast('🌐 Back online — syncing...');
  const items = await idbGetAll('pendingQueue');
  for (const item of items) {
    if (item.type === 'saveProfile')
      await window.saveProfileToCloud(item.payload);
    else if (item.type === 'addClass')
      await window.saveClassToCloud(item.payload);
    // delete from queue after successful sync
    const tx = idb.transaction('pendingQueue', 'readwrite');
    tx.objectStore('pendingQueue').delete(item.id);
  }
  if (items.length) showToast(`✅ Synced ${items.length} offline change(s)`);
});

🔄 Realtime Sync

Firestore's onSnapshot listener pushes changes to your app in real time — no polling needed. This powers live announcements, shuttle positions, and multi-device timetable sync.

Step 1 — Listen to timetable in real time

Replace your one-time getDoc call for timetable with an onSnapshot listener. Add this to your module script:

In your module script
let timetableUnsub = null; // to unsubscribe later

window.subscribeToTimetable = (uid) => {
  if (timetableUnsub) timetableUnsub(); // clean up old listener

  const ref = collection(db, 'students', uid, 'timetable');
  timetableUnsub = onSnapshot(ref, (snapshot) => {
    // Rebuild timetable object from snapshot
    window.timetable = {};
    snapshot.forEach(d => {
      const cls = { ...d.data(), _id: d.id };
      if (!window.timetable[cls.day]) window.timetable[cls.day] = [];
      window.timetable[cls.day].push(cls);
    });
    renderTT();  // your existing render function — UI updates instantly!
  });
};

// ── ANNOUNCEMENTS — push to news feed ────────────────────────
window.subscribeToAnnouncements = () => {
  const ref = collection(db, 'announcements');
  onSnapshot(ref, (snap) => {
    const list = snap.docs.map(d => d.data());
    renderAnnouncements(list); // see implementation below
  });
};

function renderAnnouncements(items) {
  const container = document.getElementById('announcements-list');
  if (!container) return;
  container.innerHTML = items
    .sort((a,b) => b.timestamp - a.timestamp)
    .map(a => `
      <div class="lrow">
        <div class="lrow-l">
          <div class="lrow-ic">📢</div>
          <div>
            <div class="lrow-t">${a.title}</div>
            <div style="font-size:.75rem;color:var(--txt3)">${a.body}</div>
          </div>
        </div>
      </div>`)
    .join('');
}

import { doc, setDoc } from 'firebase-firestore.js';

window.subscribeToShuttles = () => {

// Convert real GPS to your campus SVG coordinate space
function gpsToSVG(lat, lng) {
  // LASU approximate bounding box — measure from Google Maps
  const LAT_MIN = 6.5168, LAT_MAX = 6.5225;  // ← replace with actual values
  const LNG_MIN = 3.3645, LNG_MAX = 3.3720;
  const SVG_W   = 340, SVG_H = 220;          // ← your viewBox dimensions

  const x = ((lng - LNG_MIN) / (LNG_MAX - LNG_MIN)) * SVG_W;
  const y = ((LAT_MAX - lat) / (LAT_MAX - LAT_MIN)) * SVG_H;
  return { x, y };
}

function updateShuttleMarker(id, lat, lng, status, routeName) {
  const { x, y } = gpsToSVG(lat, lng);
  const svg  = document.getElementById('campus-svg');
  if (!svg) return;

  let marker = svg.getElementById(`shuttle-${id}`);
  if (!marker) {
    // Create new marker group
    marker = document.createElementNS('http://www.w3.org/2000/svg', 'g');
    marker.id = `shuttle-${id}`;
    marker.innerHTML = `
      <circle r="10" fill="#f0b429" opacity="0.3" class="gps-ring"/>
      <circle r="6"  fill="#f0b429" stroke="#fff" stroke-width="2"/>
      <text y="2" text-anchor="middle" font-size="5" fill="#fff">🚌</text>`;
    svg.appendChild(marker);
  }

  // Animate movement with CSS transition
  marker.style.transition = 'transform 1s ease';
  marker.setAttribute('transform', `translate(${x}, ${y})`);

  // Update status in the shuttle tab UI
  const statusEl = document.querySelector('.sh-dot');
  if (statusEl) statusEl.className = `sh-dot ${status}`;
}
⚠️
Calibrate GPS bounds carefully Open Google Maps, right-click the four corners of the LASU campus boundary and note the lat/lng. Replace the LAT_MIN/MAX and LNG_MIN/MAX constants with your measured values.

🛡️ Admin Dashboard

The admin dashboard lets LASU staff post announcements, track shuttle routes, and view registered students. It is a separate page protected by an admin custom claim.

Step 1 — Set a user as admin (one-time setup)

Run this Node.js script once to grant a specific user admin rights:

set-admin.js
const admin = require('firebase-admin');
admin.initializeApp({ credential: admin.credential.cert(require('./serviceAccountKey.json')) });

// Replace with the UID from Firebase Console → Authentication
const ADMIN_UID = 'uid_of_your_admin_account';

admin.auth()
  .setCustomUserClaims(ADMIN_UID, { admin: true })
  .then(() => console.log('Admin claim set ✓'));
Step 2 — Create admin.html
admin.html — full page structure
<!-- admin.html: Add the full Firebase module script at top -->
<script type="module">
  // ... same firebase init ...

  // Guard — redirect non-admins away
  onAuthStateChanged(auth, async (user) => {
    if (!user) { location.href = '/'; return; }
    const token = await user.getIdTokenResult();
    if (!token.claims.admin) { location.href = '/'; return; }
    initAdmin();
  });

  async function initAdmin() {
    document.getElementById('admin-panel').style.display = 'block';
    loadStudentCount();
    loadAnnouncements();
  }

  // ── POST ANNOUNCEMENT ────────────────────────
  window.postAnnouncement = async () => {
    const title = document.getElementById('ann-title').value;
    const body  = document.getElementById('ann-body').value;
    await addDoc(collection(db, 'announcements'), {
      title, body,
      timestamp: Date.now(),
      author: auth.currentUser.email
    });
    alert('✅ Announcement posted! All students will see it live.');
  };

  // ── STUDENT COUNT ────────────────────────────
  async function loadStudentCount() {
    const snap = await getDocs(collection(db, 'students'));
    document.getElementById('student-count').textContent = snap.size;
  }
</script>

CHAPTER 09

📱 Android APK Packaging

Turn your PWA into a real Android app using Bubblewrap (by Google) or PWABuilder (by Microsoft). No Android Studio required for the basic approach.

Option A — PWABuilder (easiest, browser-based)
  • Deploy your app to Firebase Hosting (free) — run firebase deploy
  • Visit pwabuilder.com and enter your live URL
  • It will score your PWA — fix any errors it shows (icon sizes, manifest)
  • Click Package for Stores → Android
  • Download the .aab file — upload to Google Play Console
Option B — Bubblewrap (developer-grade)
Terminal
# Install Bubblewrap CLI
npm install -g @bubblewrap/cli

# In your project folder, generate the Android project
bubblewrap init --manifest https://your-lasu-app.web.app/manifest.json

# Answer the prompts:
# Package name: ng.edu.lasu.navigator
# App name: LASU Navigator
# Launcher icons: provide 512x512 PNG

# Build the APK (debug)
bubblewrap build

# Output: app-release-signed.apk  ← install this on Android
Icons you must prepare

Your icons/ folder needs these files (use a tool like realfavicongenerator.net):

icons/
icon-72.png ← 72×72
icon-96.png ← 96×96
icon-128.png ← 128×128
icon-192.png ← 192×192 (required for Android home screen)
icon-512.png ← 512×512 (required for Play Store)
badge-72.png ← 72×72 monochrome (for push notification badge)
Deploy to Firebase Hosting (before packaging)
Terminal
# Install Firebase CLI
npm install -g firebase-tools

# Login and init
firebase login
firebase init hosting

# When prompted:
# Public directory: .  (your current folder)
# Single-page app: No

# Deploy
firebase deploy --only hosting

# Your app is now live at:
# https://lasu-navigator.web.app
✅
Checklist before submitting to Play Store Your PWA must score 100% on PWABuilder. It needs: a valid HTTPS URL, a manifest.json with all icon sizes, a registered Service Worker, and the display: "standalone" manifest property.
